GLOBAL RULES
------------
  - Always read payload from envelope.data  (axios → response.data.data)
  - JSON camelCase; IDs are strings ("15")
  - Soft-delete only
  - withCredentials: true (refresh cookie)
  - Authorization: Bearer <accessToken>
  - Optional: x-company-id: <activeCompanyId>

RESPONSE ENVELOPE
-----------------
  Success:
  { "success": true, "message": "...", "data": { ... }, "timestamp": "ISO" }

  Error:
  { "success": false, "message": "...", "data": null, "timestamp": "...", "path": "..." }

1. AXIOS + UNWRAP (required)
================================================================================

  // src/lib/api/axios.js
  import axios from 'axios';

  const api = axios.create({
    baseURL: process.env.NEXT_PUBLIC_API_URL, // http://localhost:3002/api/v1
    withCredentials: true,
    headers: { 'Content-Type': 'application/json' },
  });

  api.interceptors.request.use((config) => {
    const token = sessionStorage.getItem('accessToken');
    if (token) config.headers.Authorization = `Bearer ${token}`;
    const companyId = sessionStorage.getItem('activeCompanyId');
    if (companyId) config.headers['x-company-id'] = companyId;
    return config;
  });

  api.interceptors.response.use(
    (res) => res,
    async (error) => {
      const original = error.config;
      if (error.response?.status === 401 && !original._retry) {
        original._retry = true;
        try {
          const refreshed = await api.post('/auth/refresh');
          const newToken = refreshed.data.data.accessToken;
          sessionStorage.setItem('accessToken', newToken);
          original.headers.Authorization = `Bearer ${newToken}`;
          return api(original);
        } catch {
          sessionStorage.clear();
          window.location.href = '/auth/login';
        }
      }
      return Promise.reject(error);
    },
  );

  export default api;

  // unwrap
  export function unwrap(promise) {
    return promise.then((res) => res.data.data);
  }

  Method  Path                       Auth     Notes
  ------  -------------------------  -------  --------------------------------
  GET     /auth/company              Public   ?companyCode=DEMO_ACME

    "companyCode": "DEMO_ACME",
    "employeeCode": "ADMIN001",
    "password": "Admin@123"
  }

  {
    "accessToken": "...",
    "expiresIn": "15m",
    "tokenType": "Bearer",
    "requiresCompanySelection": false,
    "user": { "userId", "username", "displayName", "email" },
    "companies": [ { "companyId", "companyCode", "name", "isDefault", "status" } ],
    "activeCompany": {
      "companyId", "companyCode", "name", "status",
      "role": { "roleId", "roleCode", "roleName" },
      "subscription": { "status", "planCode", "planName", "endDate", "isCustom" },
      "modules": [
        { "moduleId", "moduleCode", "moduleName", "isActive",
          "permissions": ["view","create","edit","delete","approve"] }
      ]
    }
  }

  Side effect: sets httpOnly cookie refresh_token (Path /api/v1/auth)

GET /auth/me
------------
  Same snapshot shape + permissions[] (platform/admin nav codes when applicable)

  PLATFORM_OWNER permissions include:
    companies:view|create|edit|delete
    subscription:view|edit
    plans:view|edit
    modules:view|edit

  Product sidebar: activeCompany.modules[] only
  Admin menus: permissions[]  (companies / subscription / plans / modules)

NAV GATING
----------
  Companies menu          -> companies:view
  Subscription Plans menu -> subscription:view  (or plans:view)
  Modules menu            -> modules:view
  Company Setup           -> company ADMIN role on own tenant
