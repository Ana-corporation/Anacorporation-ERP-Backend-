POWERSHELL QUICK TEST (copy-paste — FLOW 1 + 2)
================================================================================

$base = "http://localhost:3002/api/v1"

# Login
$login = Invoke-RestMethod -Uri "$base/auth/login" -Method POST `
  -Body '{"companyCode":"DEMO_ACME","employeeCode":"ADMIN001","password":"Admin@123"}' `
  -ContentType "application/json" -SessionVariable sess

$token = $login.data.accessToken
$companyId = $login.data.activeCompany.companyId
$h = @{ Authorization = "Bearer $token" }

Write-Host "CompanyId: $companyId"

# Me
$me = Invoke-RestMethod -Uri "$base/auth/me" -Headers $h
Write-Host "Permissions count: $($me.data.permissions.Count)"

# Create branch
$branch = Invoke-RestMethod -Uri "$base/companies/$companyId/branches" -Method POST `
  -Headers $h -ContentType "application/json" `
  -Body '{"branchCode":"HQ-001","name":"Head Office","city":"Chennai","country":"India","isActive":true}'
Write-Host "Branch created: $($branch.data.name)"

# Create department
$dept = Invoke-RestMethod -Uri "$base/companies/$companyId/departments" -Method POST `
  -Headers $h -ContentType "application/json" `
  -Body '{"departmentCode":"DEPT-ADMIN","name":"Administration","isActive":true}'
Write-Host "Department created: $($dept.data.name)"
