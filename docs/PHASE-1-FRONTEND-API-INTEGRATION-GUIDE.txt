SECTION 0 — READ THIS FIRST (BACKEND CONTRACT RULES)
================================================================================

1. ALL successful API responses are WRAPPED. Always read response.data:

     {
       "success": true,
       "message": "Success",
       "data": { ... actual payload ... },
       "timestamp": "2026-07-10T06:00:00.000Z"
     }

   Axios example:
     const { data: envelope } = await axios.post('/auth/login', body);
     const payload = envelope.data;   // <-- use this

2. ALL error responses:

     {
       "success": false,
       "message": "Invalid employee code or password",
       "data": null,
       "timestamp": "...",
       "path": "/api/v1/auth/login"
     }

3. Login field is EMPLOYEE CODE (not username) — locked Phase 1 decision.
   Username exists on user record but is NOT used for login.

4. Company must be known BEFORE login:
   companyId OR companyCode is REQUIRED on POST /auth/login.
   Use GET /auth/company?companyCode= to resolve company name first.

5. refreshToken is NEVER in JSON body (stripped by backend).
   Backend sets httpOnly cookie: refresh_token
   Frontend MUST use withCredentials: true on axios.

6. Login response does NOT include flat permissions[] (admin/IAM codes).
   Call GET /auth/me after login to get permissions[] for admin menus.

7. expiresIn is a STRING (e.g. "15m"), not seconds.

8. IDs are STRINGS in JSON ("15"), except moduleId is a NUMBER.

9. roleCode from backend is uppercase (e.g. "ADMIN"), not "admin".

10. Company/subscription status can be "trial" (not only "active").

  export function unwrap(response) {
    const envelope = response.data;
    if (!envelope?.success) {
      throw new Error(envelope?.message || 'API request failed');
    }
    return envelope.data;
  }

4.1  PRE-LOGIN — RESOLVE COMPANY
------------------------------------------------------------------------------
  GET /auth/company?companyCode={code}

  Auth:     None (public)
  When:     User enters company code on login screen (before password)

  Query params:
    companyCode   string   required   e.g. "DEMO_ACME" (case-insensitive lookup)

  Response data:
    {
      "companyId": "15",
      "companyCode": "DEMO_ACME",
      "name": "Demo Acme Corp",
      "status": "trial"
    }

  Errors:
    404  Company not found

  Frontend:
    - Show company name on login form
    - Store companyId OR pass companyCode to login POST

------------------------------------------------------------------------------
4.2  SIGN UP — NEW COMPANY ADMIN
------------------------------------------------------------------------------
  POST /auth/signup

  Auth:     None (public)

  Request body:
    {
      "email": "owner@mycompany.com",
      "password": "SecurePass1",
      "confirmPassword": "SecurePass1",
      "firstName": "Owner",
      "lastName": "Name",
      "companyName": "My Company Ltd"
    }

  Password rules (backend validation):
    - Min 8 characters
    - At least 1 uppercase, 1 lowercase, 1 digit

  Response data: Same shape as login (Section 4.3)
  Cookie: refresh_token set (httpOnly)

  Notes:
    - Employee code auto-assigned: EMP-00001 style (EMP-{userId padded})
    - Default security policy row created for company
    - User gets ADMIN role with all non-platform permissions

4.3  LOGIN — MAIN ENTRY POINT
------------------------------------------------------------------------------
  POST /auth/login

  Auth:     None (public)

  Request body (companyId OR companyCode — at least one required):
    {
      "employeeCode": "EMP-00001",
      "password": "TestPass1!",
      "companyCode": "DEMO_ACME"

    OR
    {
      "employeeCode": "EMP-00001",
      "password": "TestPass1!",
      "companyId": "15"

    employeeCode   string   required
    password       string   required
    companyCode    string   optional (one of companyId/companyCode required)
    companyId      string   optional (numeric string, e.g. "15")

      "accessToken": "eyJhbGciOiJIUzI1NiIs...",
      "expiresIn": "15m",
      "tokenType": "Bearer",

        {
          "companyId": "15",
          "companyCode": "DEMO_ACME",
          "name": "Demo Acme Corp",
          "isDefault": true,
          "status": "trial"
        }
      ],
      "activeCompany": {
        "companyId": "15",
        "companyCode": "DEMO_ACME",
        "name": "Demo Acme Corp",
        "status": "trial",
        "role": {
          "roleId": "15",
          "roleCode": "ADMIN",
          "roleName": "Administrator"
        },
        "subscription": {
          "status": "trial",
          "planCode": "DEMO_STARTER",
          "planName": "Demo Starter",
          "endDate": "2027-07-10",
          "isCustom": false
        },
        "modules": [
          {
            "moduleId": 6,
            "moduleCode": "financials",
            "moduleName": "Financials",
            "isActive": true,
            "permissions": ["view", "create", "edit", "delete", "approve"]
          },
          {
            "moduleId": 16,
            "moduleCode": "crm",
            "moduleName": "CRM",
            "isActive": true,
            "permissions": ["view", "create", "edit", "delete", "approve"]
          }
        ]
      }
    }

  Cookie set by backend (do NOT read in JavaScript):
    Name:   refresh_token
    Flags:  httpOnly, secure (production), sameSite=lax
    Path:   /api/v1/auth

  NOT in response body:
    refreshToken (removed; cookie only)

  NOT in login response (call /auth/me for these):
    permissions[]   flat admin codes e.g. ["users:view", "companies:edit"]

  Frontend after login:
    1. sessionStorage.setItem('accessToken', data.accessToken)
    2. sessionStorage.setItem('authContext', JSON.stringify({
         user: data.user,
         activeCompanyId: data.activeCompany.companyId,
         companies: data.companies,
         activeCompany: data.activeCompany
       }))
    3. dispatch setAuth(...)
    4. GET /auth/me -> merge permissions[] -> hydrateIam(...)
    5. redirect to /dashboard or /company/{companyId}/dashboard

  Login error cases (HTTP status):
    400  Missing companyId/companyCode, validation error (see below)
    401  Wrong employee code/password, inactive user, locked user
    403  Company suspended/cancelled, account locked until timestamp
    404  Company not found (resolve step)

  HTTP 400 validation error example:
    {
      "success": false,
      "message": "Validation failed",
      "data": null,
      "errors": [
        { "path": ["employeeCode"], "message": "...", "code": "invalid_type" }
      ],
      "timestamp": "...",
      "path": "/api/v1/auth/login"
    }
    Map field: errors[i].path.join('.')

  HTTP 403 lockout example:
    {
      "success": false,
      "message": "Account is locked until 2026-07-10T12:30:00.000Z",
      "data": null
    }
    (No data.lockedUntil field in Phase 1 — parse from message if needed)

4.4  GET CURRENT USER + FULL SNAPSHOT
------------------------------------------------------------------------------
  GET /auth/me

  Auth:     Bearer {accessToken}

  Headers:
    Authorization: Bearer <accessToken>
    x-company-id: <companyId>   optional; if sent MUST match JWT cid or 403

  Response data:
    {
      "user": { "userId", "username", "displayName", "email" },
      "companies": [ ... ],
      "activeCompany": { ... same shape as login ... },
      "permissions": [
        "users:view",
        "users:create",
        "companies:view",
        "customers:read",
        ...
      ]
    }

  permissions[] explained:
    - Flat string codes for ADMIN/IAM/organization routes
    - Product ERP routes also accept legacy codes (customers:read -> crm:view)
    - Use for admin sidebar + settings menus
    - Product module sidebar uses activeCompany.modules[].permissions

  Frontend:
    - Call on app load if accessToken exists in sessionStorage
    - Re-hydrate Redux without re-login

  Auth:     Bearer {accessToken}

  Response data: ARRAY (not wrapped in { companies: [] })
    [
      {
        "companyId": "15",
        "companyCode": "DEMO_ACME",
        "name": "Demo Acme Corp",
        "isDefault": true,
        "status": "trial",
        "employeeCode": "EMP-00001"
      }
    ]

  Frontend:

------------------------------------------------------------------------------
4.6  SWITCH COMPANY
------------------------------------------------------------------------------
  POST /auth/switch-company

  Auth:     Bearer {accessToken}

  Request body:
    {
      "companyId": "16"
    }

  Response data: FULL login shape (new accessToken + fresh activeCompany)
    Same fields as Section 4.3 login response

  Cookie: New refresh_token issued; old session revoked server-side

  Frontend:
    1. Replace accessToken in sessionStorage
    2. Update authContext
    3. GET /auth/me (or use switch response + permissions from /me)
    4. Re-run hydrateFromSnapshot()
    5. Update activeCompanyId in auth slice
    6. Rebuild sidebar modules

------------------------------------------------------------------------------
4.7  REFRESH ACCESS TOKEN
------------------------------------------------------------------------------
  POST /auth/refresh

  Auth:     refresh_token httpOnly cookie (automatic with withCredentials)
            Optional body: { "refreshToken": "..." } if not using cookie

  Request body: empty {} or omit

  Response data: FULL snapshot (same as login — not just token)
    {
      "accessToken": "...",
      "expiresIn": "15m",
      "tokenType": "Bearer",
      "requiresCompanySelection": false,
      "user": { ... },
      "companies": [ ... ],
      "activeCompany": { ... }
    }

  Frontend (401 interceptor):
    1. POST /auth/refresh with withCredentials: true
    2. Save envelope.data.accessToken
    3. Optionally re-hydrate from full response
    4. Retry failed request with new Bearer token
    5. If refresh fails -> clearAuth() -> /auth/login

  Security note:
    Backend rotates refresh token on each refresh (reuse detection enabled)

------------------------------------------------------------------------------
4.8  LOGOUT
------------------------------------------------------------------------------
  POST /auth/logout

  Auth:     Bearer {accessToken} + refresh cookie

  Request body: empty (optional refreshToken in body if no cookie)

  Response data:
    { "message": "Logged out successfully" }

  Cookie: refresh_token cleared by backend

  Frontend:
    1. sessionStorage.removeItem('accessToken')
    2. sessionStorage.removeItem('authContext')
    3. dispatch clearAuth()
    4. redirect to /auth/login

      "role": "ADMIN",       // roleCode
      "companyId": "15",     // duplicate of cid
      "sessionId": "uuid",   // duplicate of sid
      "iat": 1783659397,
      "exp": 1783660297
    }

  Do NOT put permissions in JWT. Use snapshot + /auth/me permissions[].

SECTION 8 — AXIOS INTERCEPTORS (COPY PATTERN)
================================================================================

  // REQUEST
  api.interceptors.request.use((config) => {
    const token = sessionStorage.getItem('accessToken');
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  });

  // RESPONSE — 401 refresh
  let refreshing = false;
  api.interceptors.response.use(
    (res) => res,
    async (error) => {
      const original = error.config;
      if (error.response?.status === 401 && !original._retry) {
        original._retry = true;
        try {
          const res = await api.post('/auth/refresh');
          const data = unwrap(res);
          sessionStorage.setItem('accessToken', data.accessToken);
          original.headers.Authorization = `Bearer ${data.accessToken}`;
          return api(original);
        } catch {
          sessionStorage.clear();
          window.location.href = '/auth/login';
        }
      }
      throw error;
    }
  );

    - Logout cleanup and redirect

  // src/lib/api/auth-api.js
  import api from './axios';
  import { unwrap } from './unwrap';

  export async function resolveCompany(companyCode) {
    return unwrap(await api.get('/auth/company', { params: { companyCode } }));
  }

  export async function login({ employeeCode, password, companyCode, companyId }) {
    return unwrap(await api.post('/auth/login', {
      employeeCode, password, companyCode, companyId,
    }));
  }

  export async function signup(body) {
    return unwrap(await api.post('/auth/signup', body));
  }

  export async function getMe() {
    return unwrap(await api.get('/auth/me'));
  }

  export async function getCompanies() {
    return unwrap(await api.get('/auth/companies'));
  }

  export async function switchCompany(companyId) {
