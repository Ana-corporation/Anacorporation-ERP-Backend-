03 Access-check workflow — login → transaction

How a request is authorised, end to end.

Authenticate verify hash, MFA, not locked/expired — users + user_authentication
List tenants user_companies WHERE user_id = ? AND status = 'active'
Pick / default tenant user_companies.is_default → set company_id for the session
Resolve role user_roles WHERE user_id = ? AND company_id = ? AND is_active
Resolve permissions role_permissions → set of module + action
Apply overrides user_module_access grant/deny for this user + company
Gate by subscription company_modules.is_active AND company_subscriptions valid & not expired
Render & authorise show only permitted module + action
Every query is scoped ... WHERE company_id = :current_company_id
Effective permission = role allows action AND module active for company AND subscription valid AND NOT user_module_access = deny — an explicit grant can add a module the role lacks.

07 Runnable PostgreSQL DDL

Copy and run against Neon or local PostgreSQL. Requires PostgreSQL 14+ with pgcrypto for UUID generation.

⚠ This DDL creates a separate IAM schema. It does not replace your current organizations / Prisma tables — see Section 08 for migration mapping.
-- ERP IAM + Multi-Tenant RBAC Schema
-- PostgreSQL 14+  |  34 tables  |  Generated for iam-rbac-schema.html

CREATE EXTENSION IF NOT EXISTS "pgcrypto";

-- ─── Enums ───────────────────────────────────────────────────────────────────
CREATE TYPE company_status AS ENUM ('active','trial','suspended','closed');
CREATE TYPE membership_status AS ENUM ('active','invited','suspended');
CREATE TYPE mfa_method AS ENUM ('totp','sms','email');
CREATE TYPE module_access_type AS ENUM ('grant','deny');
CREATE TYPE subscription_status AS ENUM ('active','expired','cancelled','pending');

-- ─── Layer 0: Global ───────────────────────────────────────────────────────────
CREATE TABLE currencies (
  currency_id     UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  code            VARCHAR(3) NOT NULL UNIQUE,
  name            VARCHAR(100) NOT NULL,
  symbol          VARCHAR(8),
  decimal_places  SMALLINT NOT NULL DEFAULT 2,
  is_active       BOOLEAN NOT NULL DEFAULT true,
  created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- ─── Layer 1: Platform ─────────────────────────────────────────────────────────
CREATE TABLE super_admins (
  super_admin_id  UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  email           VARCHAR(255) NOT NULL UNIQUE,
  password_hash   TEXT NOT NULL,
  full_name       VARCHAR(200),
  is_active       BOOLEAN NOT NULL DEFAULT true,
  last_login_at   TIMESTAMPTZ,
  created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  deleted_at      TIMESTAMPTZ
);

CREATE TABLE companies (
  company_id      UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  code            VARCHAR(64) NOT NULL UNIQUE,
  legal_name      VARCHAR(255) NOT NULL,
  display_name    VARCHAR(255) NOT NULL,
  currency_id     UUID REFERENCES currencies(currency_id),
  status          company_status NOT NULL DEFAULT 'trial',
  timezone        VARCHAR(64) NOT NULL DEFAULT 'UTC',
  settings        JSONB NOT NULL DEFAULT '{}',
  created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  deleted_at      TIMESTAMPTZ
);
CREATE INDEX idx_companies_deleted ON companies(deleted_at);

CREATE TABLE subscription_plans (
  plan_id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  code            VARCHAR(64) NOT NULL UNIQUE,
  name            VARCHAR(200) NOT NULL,
  description     TEXT,
  price           NUMERIC(12,2) NOT NULL DEFAULT 0,
  billing_cycle   VARCHAR(20) NOT NULL DEFAULT 'monthly',
  max_users       INT,
  is_active       BOOLEAN NOT NULL DEFAULT true,
  created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE modules (
  module_id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  code              VARCHAR(64) NOT NULL UNIQUE,
  name              VARCHAR(200) NOT NULL,
  parent_module_id  UUID REFERENCES modules(module_id),
  sort_order        INT NOT NULL DEFAULT 0,
  is_active         BOOLEAN NOT NULL DEFAULT true,
  created_at        TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at        TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE plan_modules (
  plan_id    UUID NOT NULL REFERENCES subscription_plans(plan_id) ON DELETE CASCADE,
  module_id  UUID NOT NULL REFERENCES modules(module_id) ON DELETE CASCADE,
  PRIMARY KEY (plan_id, module_id)
);

CREATE TABLE company_subscriptions (
  subscription_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  company_id      UUID NOT NULL REFERENCES companies(company_id),
  plan_id         UUID NOT NULL REFERENCES subscription_plans(plan_id),
  status          subscription_status NOT NULL DEFAULT 'pending',
  starts_at       TIMESTAMPTZ NOT NULL,
  ends_at         TIMESTAMPTZ,
  created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE INDEX idx_company_subscriptions_company ON company_subscriptions(company_id);

CREATE TABLE company_modules (
  company_id    UUID NOT NULL REFERENCES companies(company_id) ON DELETE CASCADE,
  module_id     UUID NOT NULL REFERENCES modules(module_id) ON DELETE CASCADE,
  is_active     BOOLEAN NOT NULL DEFAULT true,
  activated_at  TIMESTAMPTZ NOT NULL DEFAULT now(),
  PRIMARY KEY (company_id, module_id)
);

CREATE TABLE company_security_policies (
  company_id              UUID PRIMARY KEY REFERENCES companies(company_id) ON DELETE CASCADE,
  min_password_length     SMALLINT NOT NULL DEFAULT 8,
  mfa_required            BOOLEAN NOT NULL DEFAULT false,
  session_timeout_min     INT NOT NULL DEFAULT 480,
  max_failed_logins       SMALLINT NOT NULL DEFAULT 5,
  password_expiry_days    INT,
  created_at              TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at              TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- ─── Layer 2: Organisation ─────────────────────────────────────────────────────
CREATE TABLE branches (
  branch_id   UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  company_id  UUID NOT NULL REFERENCES companies(company_id),
  code        VARCHAR(64) NOT NULL,
  name        VARCHAR(200) NOT NULL,
  address     JSONB NOT NULL DEFAULT '{}',
  is_active   BOOLEAN NOT NULL DEFAULT true,
  created_at  TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at  TIMESTAMPTZ NOT NULL DEFAULT now(),
  deleted_at  TIMESTAMPTZ,
  UNIQUE (company_id, code)
);
CREATE INDEX idx_branches_company ON branches(company_id) WHERE deleted_at IS NULL;

CREATE TABLE departments (
  department_id         UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  company_id            UUID NOT NULL REFERENCES companies(company_id),
  parent_department_id  UUID REFERENCES departments(department_id),
  code                  VARCHAR(64) NOT NULL,
  name                  VARCHAR(200) NOT NULL,
  created_at            TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at            TIMESTAMPTZ NOT NULL DEFAULT now(),
  deleted_at            TIMESTAMPTZ,
  UNIQUE (company_id, code)
);
CREATE INDEX idx_departments_company ON departments(company_id) WHERE deleted_at IS NULL;

CREATE TABLE designations (
  designation_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  company_id     UUID NOT NULL REFERENCES companies(company_id),
  code           VARCHAR(64) NOT NULL,
  title          VARCHAR(200) NOT NULL,
  level          SMALLINT,
  created_at     TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at     TIMESTAMPTZ NOT NULL DEFAULT now(),
  deleted_at     TIMESTAMPTZ,
  UNIQUE (company_id, code)
);
CREATE INDEX idx_designations_company ON designations(company_id) WHERE deleted_at IS NULL;

CREATE TABLE warehouses (
  warehouse_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  company_id   UUID NOT NULL REFERENCES companies(company_id),
  branch_id    UUID REFERENCES branches(branch_id),
  code         VARCHAR(64) NOT NULL,
  name         VARCHAR(200) NOT NULL,
  is_default   BOOLEAN NOT NULL DEFAULT false,
  created_at   TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at   TIMESTAMPTZ NOT NULL DEFAULT now(),
  deleted_at   TIMESTAMPTZ,
  UNIQUE (company_id, code)
);
CREATE INDEX idx_warehouses_company ON warehouses(company_id) WHERE deleted_at IS NULL;

-- ─── Layer 3: RBAC ─────────────────────────────────────────────────────────────
CREATE TABLE roles (
  role_id     UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  company_id  UUID REFERENCES companies(company_id),
  name        VARCHAR(100) NOT NULL,
  description TEXT,
  is_system   BOOLEAN NOT NULL DEFAULT false,
  created_at  TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at  TIMESTAMPTZ NOT NULL DEFAULT now(),
  deleted_at  TIMESTAMPTZ
);
CREATE UNIQUE INDEX uq_roles_company_name ON roles(company_id, name) WHERE deleted_at IS NULL;

CREATE TABLE permissions (
  permission_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  module_id     UUID NOT NULL REFERENCES modules(module_id),
  code          VARCHAR(128) NOT NULL UNIQUE,
  action        VARCHAR(64) NOT NULL,
  name          VARCHAR(200) NOT NULL,
  description   TEXT,
  created_at    TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE role_permissions (
  role_id       UUID NOT NULL REFERENCES roles(role_id) ON DELETE CASCADE,
  permission_id UUID NOT NULL REFERENCES permissions(permission_id) ON DELETE CASCADE,
  PRIMARY KEY (role_id, permission_id)
);

-- ─── Layer 4: Identity ─────────────────────────────────────────────────────────
CREATE TABLE users (
  user_id       UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  email         VARCHAR(255) NOT NULL UNIQUE,
  username      VARCHAR(100) NOT NULL UNIQUE,
  first_name    VARCHAR(100) NOT NULL,
  last_name     VARCHAR(100) NOT NULL,
  phone         VARCHAR(32),
  avatar_url    TEXT,
  is_active     BOOLEAN NOT NULL DEFAULT true,
  last_login_at TIMESTAMPTZ,
  created_at    TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at    TIMESTAMPTZ NOT NULL DEFAULT now(),
  deleted_at    TIMESTAMPTZ
);
CREATE INDEX idx_users_deleted ON users(deleted_at);

CREATE TABLE user_authentication (
  auth_id               UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id               UUID NOT NULL UNIQUE REFERENCES users(user_id) ON DELETE CASCADE,
  password_hash         TEXT NOT NULL,
  password_changed_at   TIMESTAMPTZ NOT NULL DEFAULT now(),
  failed_attempts       SMALLINT NOT NULL DEFAULT 0,
  locked_until          TIMESTAMPTZ,
  must_change_password  BOOLEAN NOT NULL DEFAULT false,
  created_at            TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at            TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE TABLE user_mfa (
  mfa_id            UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id           UUID NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
  method            mfa_method NOT NULL,
  secret_encrypted  TEXT,
  is_verified       BOOLEAN NOT NULL DEFAULT false,
  is_primary        BOOLEAN NOT NULL DEFAULT false,
  created_at        TIMESTAMPTZ NOT NULL DEFAULT now(),
  deleted_at        TIMESTAMPTZ
);
CREATE INDEX idx_user_mfa_user ON user_mfa(user_id) WHERE deleted_at IS NULL;

-- ─── Layer 5: Membership ───────────────────────────────────────────────────────
CREATE TABLE user_companies (
  membership_id   UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id         UUID NOT NULL REFERENCES users(user_id),
  company_id      UUID NOT NULL REFERENCES companies(company_id),
  employee_code   VARCHAR(64),
  department_id   UUID REFERENCES departments(department_id),
  designation_id  UUID REFERENCES designations(designation_id),
  branch_id       UUID REFERENCES branches(branch_id),
  status          membership_status NOT NULL DEFAULT 'invited',
  is_default      BOOLEAN NOT NULL DEFAULT false,
  joined_at       TIMESTAMPTZ,
  created_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  deleted_at      TIMESTAMPTZ,
  UNIQUE (user_id, company_id)
);
CREATE INDEX idx_user_companies_company ON user_companies(company_id) WHERE deleted_at IS NULL;

CREATE TABLE user_roles (
  user_role_id  UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id       UUID NOT NULL REFERENCES users(user_id),
  company_id    UUID NOT NULL REFERENCES companies(company_id),
  role_id       UUID NOT NULL REFERENCES roles(role_id),
  is_active     BOOLEAN NOT NULL DEFAULT true,
  assigned_at   TIMESTAMPTZ NOT NULL DEFAULT now(),
  assigned_by   UUID REFERENCES users(user_id)
);
CREATE INDEX idx_user_roles_lookup ON user_roles(user_id, company_id) WHERE is_active = true;

CREATE TABLE user_module_access (
  access_id    UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id      UUID NOT NULL REFERENCES users(user_id),
  company_id   UUID NOT NULL REFERENCES companies(company_id),
  module_id    UUID NOT NULL REFERENCES modules(module_id),
  access_type  module_access_type NOT NULL,
  reason       TEXT,
  expires_at   TIMESTAMPTZ,
  created_at   TIMESTAMPTZ NOT NULL DEFAULT now(),
  UNIQUE (user_id, company_id, module_id)
);

-- ─── Layer 6: Satellites ─────────────────────────────────────────────────────────
CREATE TABLE user_preferences (
  preference_id     UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id           UUID NOT NULL REFERENCES users(user_id) ON DELETE CASCADE,
  company_id        UUID REFERENCES companies(company_id),
  locale            VARCHAR(16) NOT NULL DEFAULT 'en',
  date_format       VARCHAR(32) NOT NULL DEFAULT 'YYYY-MM-DD',
  theme             VARCHAR(16) NOT NULL DEFAULT 'system',
  dashboard_layout  JSONB NOT NULL DEFAULT '{}',
  updated_at        TIMESTAMPTZ NOT NULL DEFAULT now()

CREATE TABLE user_devices (
